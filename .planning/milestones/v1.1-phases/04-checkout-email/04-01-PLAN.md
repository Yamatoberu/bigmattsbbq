    ```typescript
    import { createHash } from "crypto";

    export function newIdempotencyKey(inputs: string[]): string {
      return createHash("sha256")
        .update(inputs.sort().join("|"))
        .digest("hex")
        .slice(0, 45);
    }
    ```

    Key details:
    - Import `createHash` from Node.js built-in `crypto` module (NOT `crypto.randomUUID`)
    - Function signature changes from `()` to `(inputs: string[])`
    - Inputs are `.sort()`ed before joining with `|` separator — order-independent
    - SHA-256 hex output is 64 chars; `.slice(0, 45)` truncates to Square's max idempotency key length
    - No default export — named export `newIdempotencyKey` only
