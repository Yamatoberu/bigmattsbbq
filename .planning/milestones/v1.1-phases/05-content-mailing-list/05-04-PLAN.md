    ```typescript
    import { SignJWT, jwtVerify } from "jose";

    const TOKEN_EXPIRY = "30d";
    const ALG = "HS256";

    function getSecret(): Uint8Array {
      const secret = process.env.UNSUBSCRIBE_SECRET || process.env.BROADCAST_SECRET;
      if (!secret || secret.length < 32) {
        throw new Error(
          "Missing or too-short UNSUBSCRIBE_SECRET. Set UNSUBSCRIBE_SECRET (preferred) or BROADCAST_SECRET in .env.local to a 32+ character value."
        );
      }
      return new TextEncoder().encode(secret);
    }

    export async function signUnsubscribeToken(email: string): Promise<string> {
      if (!email) {
        throw new Error("Cannot sign unsubscribe token without an email.");
      }
      return new SignJWT({ email })
        .setProtectedHeader({ alg: ALG })
        .setIssuedAt()
        .setExpirationTime(TOKEN_EXPIRY)
        .sign(getSecret());
    }

    export async function verifyUnsubscribeToken(token: string): Promise<string> {
      const { payload } = await jwtVerify(token, getSecret(), { algorithms: [ALG] });
      const email = payload.email;
      if (typeof email !== "string" || email.length === 0) {
        throw new Error("Invalid token payload: missing email.");
      }
      return email;
    }
    ```

    ```typescript
    import { NextResponse } from "next/server";
    import { headers } from "next/headers";
    import { z } from "zod";
    import { getSupabaseClient } from "../../../lib/supabase";
    import { logError } from "../../../lib/logger";
    import { verifyUnsubscribeToken } from "../../../lib/unsubscribeToken";

    export const runtime = "nodejs";

    const schema = z.object({
      token: z.string().min(1)
    });

    export async function POST(request: Request) {
      const headerList = await headers();
      const requestId = headerList.get("x-request-id") ?? crypto.randomUUID();

      try {
        const body = await request.json();
        const parsed = schema.safeParse(body);

        if (!parsed.success) {
          return NextResponse.json(
            { error: "Missing token.", requestId },
            { status: 400 }
          );
        }

        let email: string;
        try {
          email = await verifyUnsubscribeToken(parsed.data.token);
        } catch (verifyErr) {
          logError("unsubscribe token verify failed", verifyErr, requestId);
          return NextResponse.json(
            { error: "Invalid or expired unsubscribe link.", requestId },
            { status: 401 }
          );
        }

        const supabase = getSupabaseClient();
        const { error } = await supabase
          .from("mailing_list")
          .update({ subscribed: false })
          .eq("email", email);

        if (error) {
          logError("unsubscribe update failed", error, requestId);
          return NextResponse.json(
            { error: "Unsubscribe failed. Please try again.", requestId },
            { status: 500 }
          );
        }

        return NextResponse.json({ ok: true }, { status: 200 });
      } catch (err) {
        logError("unsubscribe request failed", err, requestId);
        return NextResponse.json(
          { error: "Unsubscribe failed. Please try again.", requestId },
          { status: 500 }
        );
      }
    }
    ```

    Constraints:

    - Separate inner `try/catch` around `verifyUnsubscribeToken` so token failures return 401, not 500 (distinction matters for the client page — 401 ⇒ invalid state, 500 ⇒ generic error)
    - `.eq("email", email)` uses parameterized Supabase query — no SQL injection vector
    - Do NOT check whether the row exists first. Per D-08 enumeration prevention pattern: UPDATE affecting 0 rows is indistinguishable from UPDATE affecting 1 row in the response. An attacker with a valid token for a never-subscribed email still gets 200.

    - Route accepts POST only. Do NOT add GET handler (unsubscribe via GET is a CSRF target; POST + token-from-body is safer).
    - `verifyUnsubscribeToken` already checks expiry (exp claim) — no additional expiry logic needed here.

    ```tsx
    "use client";

    import { Suspense, useEffect, useState } from "react";
    import { useSearchParams } from "next/navigation";
    import Link from "next/link";

    type ViewState = "loading" | "success" | "invalid";

    function UnsubscribeInner() {
      const params = useSearchParams();
      const token = params.get("token");
      const [state, setState] = useState<ViewState>(token ? "loading" : "invalid");

      useEffect(() => {
        if (!token) return;
        let cancelled = false;
        fetch("/api/unsubscribe", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ token })
        })
          .then((response) => {
            if (cancelled) return;
            setState(response.ok ? "success" : "invalid");
          })
          .catch(() => {
            if (cancelled) return;
            setState("invalid");
          });
        return () => {
          cancelled = true;
        };
      }, [token]);

      return (
        <section className="section-spacing">
          <div className="mx-auto max-w-md text-center">
            <div className="glass-card p-8 mt-12">
              {state === "loading" && (
                <p className="text-sm text-smoke-700">Verifying your request…</p>
              )}

              {state === "success" && (
                <>
                  <h1
                    className="text-4xl font-semibold text-smoke-900"
                    style={{ fontFamily: "var(--font-display)" }}
                  >
                    You&apos;re unsubscribed.
                  </h1>
                  <p className="text-sm text-smoke-700 mt-4">
                    You won&apos;t receive drop notifications from us. You can rejoin anytime from the home page.
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| email link click → /unsubscribe page | Untrusted token in URL query string; verified server-side only |
| /unsubscribe page → /api/unsubscribe | Token forwarded in POST body, not query string, to avoid server logs capturing it |
| /api/unsubscribe → lib/unsubscribeToken.verifyUnsubscribeToken | JWT signature + expiry validated before DB write |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-5-01 | Spoofing | JWT forgery | mitigate | HS256 with 32+ char secret; `jwtVerify` with `algorithms: ["HS256"]` prevents algorithm confusion (alg=none, asymmetric downgrade). Tested in tests/unsubscribeToken.test.ts. |
| T-5-04-01 | Tampering | Token modification | mitigate | HMAC signature check in `jwtVerify` rejects any payload modification. Tested. |
| T-5-04-02 | Repudiation | Token replay after unsubscribe | accept | Single-use enforcement would require tracking consumed tokens. Unsubscribe is idempotent (UPDATE subscribed = false, already false → no-op), so replay causes no damage. |
| T-5-04-03 | Information Disclosure | Email enumeration via response | mitigate | Response is `{ ok: true }` with no email echo. Valid-token-unknown-email and valid-token-known-email return identical responses. DB update affecting 0 rows is indistinguishable from 1-row update from the client's perspective. |
| T-5-04-04 | Information Disclosure | Token in URL → server logs | mitigate | Client page POSTs token in request body rather than keeping it in a GET query string on the server; URL `?token=` is retained only in client-side URL bar and fetch body. Access logs on Vercel capture request path but not body. |
| T-5-04-05 | Denial of Service | Unthrottled /api/unsubscribe | accept | Low-value target; idempotent; no resource cost beyond one DB UPDATE. Revisit if abuse observed. |
| T-5-04-06 | Elevation of Privilege | Algorithm confusion attack (alg=none, alg=RS256 with pub key as secret) | mitigate | `algorithms: ["HS256"]` in jwtVerify pins to one algorithm; jose rejects any token with a different alg header. |
| T-5-04-07 | Information Disclosure | Weak/leaked BROADCAST_SECRET → token forgery | mitigate | Plan 01 requires 32+ char cryptographically-random secret; lib/unsubscribeToken.getSecret() throws if `< 32` chars. `.env.local` is gitignored. |
</threat_model>
