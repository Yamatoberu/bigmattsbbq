    ```typescript
    import { NextResponse } from "next/server";
    import { headers } from "next/headers";
    import { z } from "zod";
    import { Resend } from "resend";
    import { getSupabaseClient } from "../../../../lib/supabase";
    import { logError } from "../../../../lib/logger";
    import { signUnsubscribeToken } from "../../../../lib/unsubscribeToken";

    export const runtime = "nodejs";

    const schema = z.object({
      subject: z.string().min(1).max(200),
      html: z.string().min(1),
      dropId: z.string().optional()
    });

    function getBaseUrl(headerList: Headers): string {
      const envUrl = process.env.NEXT_PUBLIC_SITE_URL;
      if (envUrl) return envUrl.replace(/\/$/, "");
      const forwardedProto = headerList.get("x-forwarded-proto") ?? "https";
      const host = headerList.get("host") ?? "bigmattsbbq.com";
      return `${forwardedProto}://${host}`;
    }

    function authorize(headerList: Headers): boolean {
      const secret = process.env.BROADCAST_SECRET;
      if (!secret || secret.length < 16) return false;
      const authHeader = headerList.get("authorization");
      if (!authHeader) return false;
      return authHeader === `Bearer ${secret}`;
    }

    export async function POST(request: Request) {
      const headerList = await headers();
      const requestId = headerList.get("x-request-id") ?? crypto.randomUUID();

      if (!authorize(headerList)) {
        return NextResponse.json(
          { error: "Unauthorized" },
          { status: 401 }
        );
      }

      try {
        const body = await request.json();
        const parsed = schema.safeParse(body);

        if (!parsed.success) {
          return NextResponse.json(
            { error: "Invalid broadcast payload.", requestId },
            { status: 400 }
          );
        }

        const { subject, html, dropId } = parsed.data;

        const supabase = getSupabaseClient();
        const { data: subscribers, error: listErr } = await supabase
          .from("mailing_list")
          .select("email")
          .eq("subscribed", true);

        if (listErr) {
          logError("broadcast subscriber fetch failed", listErr, requestId);
          return NextResponse.json(
            { error: "Broadcast failed.", requestId },
            { status: 500 }
          );
        }

        const list: Array<{ email: string }> = subscribers ?? [];
        if (list.length === 0) {
          return NextResponse.json(
            { sent: 0, failed: 0, requestId },
            { status: 200 }
          );
        }

        const resendKey = process.env.RESEND_API_KEY;
        if (!resendKey) {
          logError("broadcast missing RESEND_API_KEY", new Error("RESEND_API_KEY not set"), requestId);
          return NextResponse.json(
            { error: "Broadcast failed.", requestId },
            { status: 500 }
          );
        }
        const resend = new Resend(resendKey);
        const from = process.env.EMAIL_FROM ?? "Big Matt's BBQ <orders@bigmattsbbq.com>";
        const baseUrl = getBaseUrl(headerList);
        const template = dropId ? `drop_notification:${dropId}` : "drop_notification";

        let sent = 0;
        let failed = 0;

        for (const subscriber of list) {
          let resendId: string | null = null;
          let status: "sent" | "failed" = "sent";
          try {
            const token = await signUnsubscribeToken(subscriber.email);
            const unsubscribeUrl = `${baseUrl}/unsubscribe?token=${encodeURIComponent(token)}`;
            const finalHtml = `${html}
<hr style="margin-top: 24px; border: 0; border-top: 1px solid #ccc;" />
<p style="font-size: 12px; color: #666;">
  Don't want these? <a href="${unsubscribeUrl}">Unsubscribe</a>.
</p>`;

            const { data, error: sendErr } = await resend.emails.send({
              from,
              to: subscriber.email,
              subject,
              html: finalHtml
            });

            if (sendErr) {
              status = "failed";
              logError("broadcast send failed for recipient", sendErr, requestId);
            } else {
              resendId = data?.id ?? null;
              sent += 1;
            }
          } catch (sendCatch) {
            status = "failed";
            logError("broadcast send threw for recipient", sendCatch, requestId);
          }

          if (status === "failed") failed += 1;

          const { error: logErr } = await supabase.from("email_logs").insert({
            recipient: subscriber.email,
            template,
            status,
            resend_id: resendId
          });
          if (logErr) {
            logError("broadcast email_logs insert failed", logErr, requestId);
          }
        }

        return NextResponse.json({ sent, failed, requestId }, { status: 200 });
      } catch (err) {
        logError("broadcast request failed", err, requestId);
        return NextResponse.json(
          { error: "Broadcast failed.", requestId },
          { status: 500 }
        );
      }
    }
    ```

    Constraints:
    - Named exports only (POST, runtime). No default export.
    - Bearer check runs BEFORE body parsing — no info leaked to unauthorized clients via Zod errors or logs
    - Resend is instantiated INSIDE the handler (per RESEARCH Pitfall 4 — not at module level, because RESEND_API_KEY might be missing at build time and would throw)
    - email_logs insert runs for EVERY recipient regardless of send success/failure — captures the attempt
    - Per RESEARCH Pitfall 5: no drop_id column. Encode dropId in `template` string as `drop_notification:{dropId}`, or just `drop_notification` if absent
    - Unsubscribe link URL uses `encodeURIComponent` on the token to handle any special characters (JWTs are base64url so usually safe, but belt-and-suspenders)
    - Base URL derived from `NEXT_PUBLIC_SITE_URL` env if set, otherwise from request headers (`x-forwarded-proto`, `host`) — works both in local dev and Vercel production
    - Response body never echoes subscriber emails — only counts
    - Iteration loop is sequential (not `Promise.all`) to avoid Resend rate limit bursts; MVP list is small (assumption A2 in RESEARCH)
    - `from` address comes from `EMAIL_FROM` env var (already populated in .env.local), not hardcoded
    - EVERY log statement uses `logError` with `requestId` — never `console.log` in route code
    - Do NOT return 401 with `requestId` in the body — keep unauthorized response minimal (`{ error: "Unauthorized" }`) so scanners/enumeration attempts get no extra metadata
    - Zod schema caps subject at 200 chars — prevents oversized email headers

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| External caller (curl / Matt) → /api/admin/broadcast | Any internet client can hit this path; bearer header is the sole gate |
| Route → Resend API | Outbound to 3rd-party service over HTTPS; RESEND_API_KEY authenticates us |
| Route → Supabase | Service-role key; RLS bypassed; reads mailing_list, writes email_logs |
| Broadcast email → recipient inbox | Recipient can click unsubscribe link; token verified server-side |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-5-02 | Spoofing | Unauthorized broadcast trigger | mitigate | Bearer check runs FIRST; `authorize()` returns false for missing/short/wrong secret and for missing `Authorization` header. All 4 auth tests in tests/broadcast.test.ts cover this explicitly. Without the secret, attackers cannot send emails at all. |
| T-5-05-01 | Information Disclosure | Subscriber email enumeration via response body | mitigate | Response body contains only `{ sent, failed, requestId }` — no subscriber emails echoed. 401 responses contain only `{ error: "Unauthorized" }` — no request metadata. |
| T-5-05-02 | Tampering | Secret comparison timing attack | accept | String equality in Node.js is not constant-time, but BROADCAST_SECRET is 64 hex chars (256 bits of entropy); brute-forcing via timing across internet latency is infeasible. Constant-time compare (crypto.timingSafeEqual) could be added later. |
| T-5-05-03 | Repudiation | No audit trail for broadcast trigger | mitigate | Every send writes one email_logs row with recipient + status + timestamp (created_at default); `requestId` appears in logError calls for cross-referencing |
| T-5-05-04 | Denial of Service | Large mailing list exhausts Resend budget or times out | accept | MVP list is small (assumption A2). Sequential loop prevents rate-limit bursts. Vercel serverless timeout is 10s–60s depending on plan — if list grows beyond ~100, migrate to `resend.batch.send` (documented in RESEARCH Alternatives Considered). |
| T-5-05-05 | Elevation of Privilege | Secret leakage via error response | mitigate | No error path echoes `process.env.BROADCAST_SECRET`. `logError` writes to server logs only (console.error via logger.ts). |
| T-5-05-06 | Information Disclosure | Missing RESEND_API_KEY returns 500 but attacker already authorized | accept | If attacker has valid BROADCAST_SECRET, they're trusted; RESEND_API_KEY absence is an ops issue, not a security boundary |
| T-5-05-07 | Tampering | Base URL spoofing via Host header | mitigate | `NEXT_PUBLIC_SITE_URL` env var takes precedence when set; fallback uses request headers only for localhost dev. Production deploy MUST set NEXT_PUBLIC_SITE_URL. |
| T-5-05-08 | Spoofing | Email display name forgery | accept | Standard email concern; `EMAIL_FROM` env var controls sender identity; Resend domain verification (configured outside code) is the mitigation |
</threat_model>
