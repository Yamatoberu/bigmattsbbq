   ```ts
   const bundleVariationIdToName = useMemo(() => {
     const map = new Map<string, string>();
     for (const pkg of PACKAGES) {
       if (pkg.bundleVariationId) {
         map.set(pkg.bundleVariationId, pkg.name);
       }
     }
     return map;
   }, []);
   ```
   Notes:
   - `PACKAGES` is a module-level constant, so the dependency array is `[]`.
   - Do not create a new file or new module — keep this co-located with the other memoized lookups.

   ```ts
   const cartDetails = items.map((item) => {
   ```
